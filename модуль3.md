Ниже — Модуль 3. Он использует уже существующие модели и **не требует изменения схемы сохранения**.

### Принятые правила

Поскольку точных формул боя и таблиц монстров в GDD нет, их баланс вынесен в явные константы:

- фиксированный шаг симуляции — **50 мс**;
- снижение физического урона: `100 / (100 + armor)`;
- магический урон учитывает 50% брони;
- i-frames героя — **180 мс**, независимо от состояния анимации;
- барьер отряда — 20% суммарного максимального HP;
- босс — каждая десятая волна, таймер боя — 45 игровых секунд;
- четыре акта по 30 волн, после них — бесконечная Цитадель;
- поражение не отнимает золото: через три секунды повторяется текущая волна.

Навыки из Модуля 2 получают исполняемую семантику: урон, DoT, лечение, призыв, щиты и временные модификаторы. Численные параметры призыва и лечения — стартовый баланс, не утверждения о наличии этих чисел в GDD.

**Архитектура:** движок не знает о виджетах. Театр рисует снимки состояния; монстры и всплывающие цифры — обычные Python-объекты, а не динамические `QWidget`. Это устраняет основной источник ошибок с удалёнными Qt-объектами.

```python
    enemy_id: str
    template_id: str
    name: str

    hp: float
    damage: float
    armor: float

    speed: float

    y: float

        return self.rank == EnemyRank.BOSS

    @property

    @property

def act_for_wave(wave: int) -> Act:
    if type(wave) is not int or wave < 1:
        raise ValueError("Номер волны должен быть целым >= 1")
    index = min((wave - 1) // WAVES_PER_ACT, len(Act) - 1)
    return tuple(Act)[index]


def spawn_wave(
    wave: int,
    rng: random.Random | None = None,
    *,
    generation: int = 0,
) -> list[Enemy]:
    act = act_for_wave(wave)
    generator = rng if rng is not None else random.Random()

    boss_wave = wave % BOSS_EVERY == 0
    count = 1 if boss_wave else min(6, 3 + (wave - 1) // 20)

    # Общая прогрессия сложности, без сброса при переходе акта.
    hp_scale = (1.0 + 0.09 * (wave - 1)) ** 1.35
    damage_scale = (1.0 + 0.06 * (wave - 1)) ** 1.15

    result = []

    for index in range(count):
        template = generator.choice(BESTIARY[act])

        if boss_wave:
            rank = EnemyRank.BOSS
            affixes = ()
        elif generator.random() < min(0.35, 0.12 + wave * 0.002):
            rank = EnemyRank.ELITE
            amount = 2 if wave >= 40 else 1
            affixes = tuple(generator.sample(tuple(EnemyAffix), amount))
        else:
            rank = EnemyRank.NORMAL
            affixes = ()

        hp_multiplier = {
            EnemyRank.NORMAL: 1.0,
            EnemyRank.ELITE: 1.9,
            EnemyRank.BOSS: 7.0,
        }[rank]

        attack_multiplier = {
            EnemyRank.NORMAL: 1.0,
            EnemyRank.ELITE: 1.35,
            EnemyRank.BOSS: 1.8,
        }[rank]

        armor = template.armor + wave * 0.7
        interval = template.attack_interval
        speed = template.speed

        if EnemyAffix.STONE_SKIN in affixes:
            armor = armor * 2 + 25

        if EnemyAffix.EXTRA_FAST in affixes:
            interval *= 0.70
            speed *= 1.55

        max_hp = template.hp * hp_scale * hp_multiplier

        result.append(Enemy(
            enemy_id=f"enemy-{generation}-{wave}-{index}",
            template_id=template.template_id,
            name=BOSS_NAMES[act] if boss_wave else template.name,
            act=act,
            rank=rank,
            affixes=affixes,
            max_hp=max_hp,
            hp=max_hp,
            damage=template.damage * damage_scale * attack_multiplier,
            armor=armor,
            attack_interval=interval,
            speed=speed,
            magical=template.magical,
            x=460.0 + index * 48,
            y=99.0 + (index % 2) * 12,
        ))

    return result
```

# 3. Боевой движок

Главные особенности:

- `advance(dt)` не зависит от FPS;
- кулдауны принадлежат конкретному герою;
- i-frames не меняют анимацию автоматически;
- повторный `ATTACK` не перезапускает уже идущую анимацию;
- награда выдаётся единожды при переходе в межволновую фазу;
- изменения моделей героев и рун подхватываются без обнуления кулдаунов;
- активные награды используют `FarmProfile` из сохранения;
- накопленный частичный прогресс волны сохраняется через `flush()`.

```python
@dataclass(frozen=True)
class Hit:
    damage: float
    critical: bool = False
    dodged: bool = False


def roll_damage(
    power: float,
    armor: float,
    *,
    magical: bool = False,
    crit_chance: float = 0.0,
    crit_multiplier: float = 1.5,
    evasion: float = 0.0,
    rng: random.Random | None = None,
) -> Hit:
    values = (power, armor, crit_chance, crit_multiplier, evasion)
    if not all(math.isfinite(value) for value in values):
        raise ValueError("Параметры урона должны быть конечными")
    if power < 0 or armor < 0 or crit_multiplier < 1:
        raise ValueError("Некорректные параметры урона")
    if not 0 <= crit_chance <= 1 or not 0 <= evasion <= 1:
        raise ValueError("Вероятность должна находиться в [0, 1]")

    generator = rng if rng is not None else random.Random()

    if generator.random() < evasion:
        return Hit(0.0, dodged=True)

    critical = generator.random() < crit_chance
    effective_armor = armor * (0.5 if magical else 1.0)

    damage = power * 100.0 / (100.0 + effective_armor)
    if critical:
        damage *= crit_multiplier

    return Hit(damage, critical=critical)

class CombatManager(QObject):
    frame_changed = Signal(object)
    damage_event = Signal(object)
    skill_used = Signal(str, str)
    wave_finished = Signal(object)
    party_defeated = Signal(object)

    def __init__(
        self,
        state: GameState,
        *,
        rng: random.Random | None = None,
        parent: QObject | None = None,
    ) -> None:
        super().__init__(parent)
        self.state = state
        self.rng = rng if rng is not None else random.Random()
        self.runes = RunesTree()

        self.running = False
        self.phase = "stopped"
        self.time = 0.0

        self._accumulator = 0.0
        self._phase_left = 0.0
        self._generation = 0
        self._wave = state.data.current_wave
        self._wave_elapsed = 0.0
        self._wave_real_elapsed = 0.0
        self._wave_total_hp = 1.0
        self._continued_wave = False

        self._heroes: dict[str, HeroRuntime] = {}
        self._enemies: list[Enemy] = []
        self._dots: list[DamageOverTime] = []

        self.barrier = 0.0
        self.barrier_max = 0.0
        self._effects = self.runes.effects(state.data.unlocked_runes)

    @property
    def heroes(self) -> tuple[HeroRuntime, ...]:
        return tuple(self._heroes.values())

    @property
    def enemies(self) -> tuple[Enemy, ...]:
        return tuple(self._enemies)

    def start(self) -> None:
        self.state.assert_owner_thread()
        if self.running:
            return
        if not 3 <= len(self.state.data.party) <= 4:
            raise ValueError("Боевой отряд должен содержать 3–4 героев")

        self.running = True
        self._accumulator = 0.0
        self._sync_heroes()
        self._spawn()
        self.frame_changed.emit(self.frame())

    def stop(self) -> None:
        self.state.assert_owner_thread()
        if self.running:
            self.flush()
        self.running = False
        self.phase = "stopped"

    def _sync_heroes(self) -> None:
        """Обновить модели, сохраняя HP-пропорцию, ману и текущие кулдауны."""
        members = self.runes.squad(self.state.data)
        if not 3 <= len(members) <= 4:
            raise ValueError("Боевой отряд должен содержать 3–4 героев")

        snapshots = {hero.hero_id: hero for hero in self.state.data.party}
        updated = {}

        for index, member in enumerate(members):
            hero_id = member.hero.hero_id
            old = self._heroes.get(hero_id)

            if old is None:
                snapshot = snapshots[hero_id]
                runtime = HeroRuntime(
                    hero=member.hero,
                    stats=member.stats,
                    hp=member.stats.max_hp * snapshot.hp / snapshot.max_hp,
                    mana=float(member.stats.max_mana),
                    x=HERO_POSITIONS[index][0],
                    y=HERO_POSITIONS[index][1],
                )
            else:
                hp_ratio = old.hp / old.stats.max_hp
                old.hero = member.hero
                old.stats = member.stats
                old.hp = max(0.0, min(member.stats.max_hp,
                                      member.stats.max_hp * hp_ratio))
                old.mana = min(old.mana, member.stats.max_mana)
                old.x, old.y = HERO_POSITIONS[index]
                runtime = old

            for skill in member.hero.skills:
                runtime.skill_cooldowns.setdefault(skill.skill_id, 0.0)

            updated[hero_id] = runtime

        old_max = self.barrier_max
        self._heroes = updated
        self._effects = self.runes.effects(self.state.data.unlocked_runes)

        self.barrier_max = BARRIER_RATIO * sum(
            hero.stats.max_hp for hero in self._heroes.values()
        )
        self.barrier = (
            self.barrier_max
            if old_max == 0
            else min(self.barrier_max, self.barrier * self.barrier_max / old_max)
        )

    def _spawn(self) -> None:
        self._sync_heroes()
        self._generation += 1
        self._wave = self.state.data.current_wave

        self._enemies = spawn_wave(
            self._wave,
            self.rng,
            generation=self._generation,
        )
        self._wave_total_hp = sum(enemy.max_hp for enemy in self._enemies)

        partial = float(Fraction(self.state.data.wave_fraction))
        self._continued_wave = partial > 0

        # Сохранение содержит агрегат прогресса, а не точное состояние мобов.
        for enemy in self._enemies:
            enemy.hp = max(0.001, enemy.max_hp * (1.0 - partial))

        self._dots.clear()
        self._wave_elapsed = 0.0
        self._wave_real_elapsed = 0.0
        self.phase = "battle"

    def advance(self, seconds: float) -> None:
        self.state.assert_owner_thread()

        if not math.isfinite(seconds) or not 0 <= seconds <= 60:
            raise ValueError("advance принимает от 0 до 60 секунд")
        if not self.running:
            return

        self._sync_heroes()

        speed = 1.0 + self._effects.active_speed_bonus
        self._accumulator += seconds * speed

        while self._accumulator + 1e-12 >= FIXED_STEP:
            self._accumulator = max(0.0, self._accumulator - FIXED_STEP)
            self._step(FIXED_STEP, FIXED_STEP / speed)

        self.frame_changed.emit(self.frame())

    @staticmethod
    def _animate(actor, state: str, duration: float) -> None:
        if actor.animation == "die":
            return

        # Удар не перебивает каст или уже начатую атаку.
        if state == "hurt" and actor.animation in ("attack", "cast"):
            return

        # Не перезапускаем одну и ту же анимацию.
        if actor.animation == state:
            return

        actor.animation = state
        actor.animation_left = duration

    def _tick_animations(self, dt: float) -> None:
        for actor in (*self._heroes.values(), *self._enemies):
            if actor.animation == "die":
                continue

            if actor.animation_left > 0:
                actor.animation_left = max(0.0, actor.animation_left - dt)
                if actor.animation_left == 0:
                    actor.animation = "idle"

    def _step(self, dt: float, real_dt: float) -> None:
        self.time += dt
        self._tick_animations(dt)

        # Кулдауны продолжают идти и между волнами.
        for hero in self._heroes.values():
            hero.basic_timer = max(0.0, hero.basic_timer - dt)
            for skill_id in hero.skill_cooldowns:
                hero.skill_cooldowns[skill_id] = max(
                    0.0, hero.skill_cooldowns[skill_id] - dt
                )
            if hero.alive:
                hero.mana = min(
                    hero.stats.max_mana,
                    hero.mana + (3 + hero.stats.max_mana * 0.01) * dt,
                )

        if self.phase != "battle":
            self._phase_left -= dt
            if self._phase_left <= 0:
                if self.phase == "defeat":
                    for hero in self._heroes.values():
                        hero.hp = float(hero.stats.max_hp)
                        hero.mana = float(hero.stats.max_mana)
                        hero.invulnerable_until = 0.0
                        hero.animation = "idle"
                        hero.animation_left = 0.0
                        hero.buffs.clear()
                    self.barrier = self.barrier_max
                    self.flush()
                else:
                    self.barrier = min(
                        self.barrier_max,
                        self.barrier + self.barrier_max * 0.25,
                    )
                self._spawn()
            return

        self._wave_elapsed += dt
        self._wave_real_elapsed += real_dt

        self._tick_dots(dt)
        self._tick_heroes(dt)

        if not any(hero.alive for hero in self._heroes.values()):
            self._defeat("Отряд пал")
            return

        if not any(enemy.alive for enemy in self._enemies):
            self._finish_wave()
            return

        self._tick_enemies(dt)

        if not any(hero.alive for hero in self._heroes.values()):
            self._defeat("Отряд пал")
            return

        if (
            any(enemy.is_boss and enemy.alive for enemy in self._enemies)
            and self._wave_elapsed >= BOSS_TIME_LIMIT
        ):
            self._defeat("Истекло время боя с боссом")

    def _target(self) -> Enemy | None:
        return min(
            (enemy for enemy in self._enemies if enemy.alive),
            key=lambda enemy: enemy.x,
            default=None,
        )

    def _tick_heroes(self, dt: float) -> None:
        for hero in tuple(self._heroes.values()):
            if not hero.alive:
                continue

            target = self._target()
            if target is None:
                break

            if hero.summon_left > 0:
                hero.summon_left = max(0.0, hero.summon_left - dt)
                hero.summon_timer -= dt
                if hero.summon_timer <= 0:
                    hero.summon_timer += 1.0
                    self._hit_enemy(hero, target, hero.summon_power, True)

            target = self._target()
            if target is None:
                break

            # Боевой театр стилизован: это дистанция доступности цели,
            # а не физическая коллизия спрайтов.
            ranged = hero.hero.hero_class in (
                HeroClass.PYROMANCER,
                HeroClass.RANGER,
                HeroClass.NECROMANCER,
            )
            reach = 440 if ranged else 275
            if target.x > reach:
                continue

            if hero.animation in ("attack", "cast", "hurt"):
                continue

            cast = False
            for skill in hero.hero.skills:
                cost = self._skill_cost(hero, skill)
                if (
                    hero.skill_cooldowns[skill.skill_id] <= 1e-9
                    and hero.mana >= cost
                    and self._cast(hero, skill)
                ):
                    hero.mana -= cost
                    hero.skill_cooldowns[skill.skill_id] = skill.cooldown_seconds
                    self._animate(hero, "cast", 0.45)
                    self.skill_used.emit(hero.hero.hero_id, skill.skill_id)
                    cast = True
                    break

            if not cast and hero.basic_timer <= 1e-9:
                magical = hero.hero.definition.magical
                power = (
                    hero.stats.magical_damage
                    if magical else hero.stats.physical_damage
                )
                self._animate(hero, "attack", 0.35)
                hero.basic_timer = BASIC_INTERVALS[hero.hero.hero_class]
                self._hit_enemy(hero, target, power, magical)

    def _skill_cost(self, hero: HeroRuntime, skill: Skill) -> float:
        berserk = (
            "berserker" in self._effects.keystones
            and hero.hp < hero.stats.max_hp * 0.40
        )
        return skill.mana_cost * (2 if berserk else 1)

    def _cast(self, hero: HeroRuntime, skill: Skill) -> bool:
        target = self._target()
        if target is None:
            return False

        effect = skill.effect
        magical = hero.hero.definition.magical
        base = (
            hero.stats.magical_damage
            if magical else hero.stats.physical_damage
        )

        if effect == "heal":
            allies = [entry for entry in self._heroes.values() if entry.alive]
            ally = min(allies, key=lambda entry: entry.hp / entry.stats.max_hp)
            if ally.hp >= ally.stats.max_hp * 0.90:
                return False
            self._heal(hero=ally, amount=hero.stats.magical_damage
                       * skill.power_multiplier)
            return True

        if effect == "damage_reduction":
            hero.buffs["reduction"] = (
                self.time + skill.duration_seconds,
                skill.power_multiplier,
            )
            return True

        if effect == "evasion_buff":
            hero.buffs["evasion"] = (
                self.time + skill.duration_seconds,
                skill.power_multiplier,
            )
            return True

        if effect == "party_damage_buff":
            for ally in self._heroes.values():
                ally.buffs["damage"] = (
                    self.time + skill.duration_seconds,
                    skill.power_multiplier,
                )
            return True

        if effect == "max_hp_shield":
            # Барьер общий. Не позволяем накопить его выше штатного максимума.
            if self.barrier >= self.barrier_max * 0.90:
                return False
            amount = sum(
                entry.stats.max_hp
                for entry in self._heroes.values() if entry.alive
            ) * skill.power_multiplier
            self.barrier = min(self.barrier_max, self.barrier + amount)
            return True

        if effect == "damage_taken_debuff":
            target.marked_until = self.time + skill.duration_seconds
            target.marked_bonus = skill.power_multiplier
            return True

        if effect == "summon":
            hero.summon_left = skill.duration_seconds
            hero.summon_timer = 0.0
            hero.summon_power = hero.stats.magical_damage * skill.power_multiplier
            return True

        targets = (
            [enemy for enemy in self._enemies if enemy.alive]
            if skill.target == "all_enemies"
            else [target]
        )

        if effect in ("poison_per_second", "burn_per_second"):
            for enemy in targets:
                self._dots.append(DamageOverTime(
                    owner_id=hero.hero.hero_id,
                    enemy_id=enemy.enemy_id,
                    power=base * skill.power_multiplier,
                    remaining=skill.duration_seconds,
                    magical=effect == "burn_per_second",
                ))
            return True

        if effect in ("damage", "life_drain"):
            for enemy in targets:
                if not hero.alive:
                    break
                dealt = self._hit_enemy(
                    hero, enemy, base * skill.power_multiplier, magical
                )
                if effect == "life_drain" and hero.alive:
                    self._heal(hero, dealt * 0.35)
            return True

        raise ValueError(f"Не реализован эффект навыка: {effect}")

    def _tick_dots(self, dt: float) -> None:
        enemies = {enemy.enemy_id: enemy for enemy in self._enemies}
        remaining = []

        for dot in self._dots:
            owner = self._heroes.get(dot.owner_id)
            enemy = enemies.get(dot.enemy_id)

            if owner is None or enemy is None or not enemy.alive:
                continue

            dot.remaining -= dt
            dot.timer -= dt

            if dot.timer <= 1e-9:
                dot.timer += 1.0
                self._hit_enemy(
                    owner, enemy, dot.power, dot.magical,
                    allow_crit=False, allow_splash=False,
                )

            if dot.remaining > 1e-9 and enemy.alive:
                remaining.append(dot)

        self._dots = remaining

    def _hit_enemy(
        self,
        hero: HeroRuntime,
        enemy: Enemy,
        power: float,
        magical: bool,
        *,
        allow_crit: bool = True,
        allow_splash: bool = True,
    ) -> float:
        if not enemy.alive:
            return 0.0

        power *= 1.0 + hero.buff("damage", self.time)

        if (
            "berserker" in self._effects.keystones
            and hero.hp < hero.stats.max_hp * 0.40
        ):
            power *= 1.8

        if enemy.marked_until > self.time:
            power *= 1.0 + enemy.marked_bonus

        crit_multiplier = hero.stats.crit_multiplier
        if magical and "supernova" in self._effects.keystones:
            crit_multiplier *= 1.5

        hit = roll_damage(
            power,
            enemy.armor,
            magical=magical,
            crit_chance=hero.stats.crit_chance if allow_crit else 0.0,
            crit_multiplier=crit_multiplier,
            rng=self.rng,
        )

        dealt = min(enemy.hp, hit.damage)
        enemy.hp = max(0.0, enemy.hp - dealt)

        self.damage_event.emit(DamageEvent(
            enemy.enemy_id, enemy.x, enemy.y - 32,
            dealt, "magical" if magical else "physical", hit.critical,
        ))

        if enemy.is_boss:
            enemy.rage = min(
                enemy.rage_max,
                enemy.rage + dealt / enemy.max_hp * 30,
            )

        if not enemy.alive:
            enemy.animation = "die"
            enemy.animation_left = 0.0
        else:
            self._animate(enemy, "hurt", 0.20)

        if hero.alive:
            self._heal(
                hero,
                dealt * BASE_LIFESTEAL[hero.hero.hero_class],
            )

        if EnemyAffix.REFLECTIVE in enemy.affixes and dealt > 0 and hero.alive:
            # Отражение не отражается обратно.
            self._hit_hero(hero, dealt * 0.12, True)

        if (
            allow_splash
            and not magical
            and "titan_sweep" in self._effects.keystones
            and hero.alive
        ):
            neighbor = next(
                (
                    entry for entry in self._enemies
                    if entry.alive
                    and entry is not enemy
                    and abs(entry.x - enemy.x) <= 80
                ),
                None,
            )
            if neighbor:
                self._hit_enemy(
                    hero, neighbor, power * 0.40, False,
                    allow_crit=False, allow_splash=False,
                )

        return dealt

    def _hit_hero(
        self,
        hero: HeroRuntime,
        power: float,
        magical: bool,
    ) -> float:
        if not hero.alive or self.time < hero.invulnerable_until:
            return 0.0

        hit = roll_damage(
            power,
            hero.stats.armor,
            magical=magical,
            evasion=min(
                0.95,
                hero.stats.evasion + hero.buff("evasion", self.time),
            ),
            rng=self.rng,
        )

        # Даже уворот потребляет короткое окно защиты от повторной коллизии.
        hero.invulnerable_until = self.time + HERO_IFRAMES

        if hit.dodged:
            self.damage_event.emit(DamageEvent(
                hero.hero.hero_id, hero.x, hero.y - 32,
                0, "dodge",
            ))
            return 0.0

        damage = hit.damage * (
            1.0 - min(0.90, hero.buff("reduction", self.time))
        )

        absorbed = min(self.barrier, damage)
        self.barrier -= absorbed
        damage -= absorbed

        if "ether_shield" in self._effects.keystones:
            mana_absorb = min(hero.mana, damage * 0.30)
            hero.mana -= mana_absorb
            damage -= mana_absorb

        dealt = min(hero.hp, damage)
        hero.hp = max(0.0, hero.hp - dealt)

        self.damage_event.emit(DamageEvent(
            hero.hero.hero_id, hero.x, hero.y - 32,
            dealt,
            ("magical" if magical else "physical") if dealt > 0 else "barrier",
        ))

        if not hero.alive:
            hero.animation = "die"
            hero.animation_left = 0.0
        elif dealt > 0:
            self._animate(hero, "hurt", 0.20)

        if (
            hero.alive
            and "second_wind" in self._effects.keystones
            and hero.hp < hero.stats.max_hp * 0.20
            and self.time >= hero.second_wind_ready
        ):
            hero.second_wind_ready = self.time + 60
            self._heal(hero, hero.stats.max_hp * 0.30)

        return dealt

    def _heal(self, hero: HeroRuntime, amount: float) -> None:
        if not hero.alive or amount <= 0:
            return

        restored = min(hero.stats.max_hp - hero.hp, amount)
        if restored <= 0:
            return

        hero.hp += restored
        self.damage_event.emit(DamageEvent(
            hero.hero.hero_id, hero.x, hero.y - 32,
            restored, "heal",
        ))

    def _tick_enemies(self, dt: float) -> None:
        for index, enemy in enumerate(self._enemies):
            if not enemy.alive:
                enemy.corpse_left -= dt
                continue

            if enemy.is_boss:
                enemy.rage = min(
                    enemy.rage_max,
                    enemy.rage + dt * enemy.rage_max / 30.0,
                )

            enemy.attack_timer = max(0.0, enemy.attack_timer - dt)
            destination = 225.0 + (index % 3) * 20

            if enemy.x > destination:
                enemy.x = max(destination, enemy.x - enemy.speed * dt)
                if enemy.animation in ("idle", "run"):
                    enemy.animation = "run"
                continue

            if enemy.animation == "run":
                enemy.animation = "idle"

            allies = [hero for hero in self._heroes.values() if hero.alive]
            if not allies:
                return

            if enemy.attack_timer <= 1e-9:
                target = (
                    self.rng.choice(allies)
                    if enemy.magical else allies[0]
                )

                self._animate(enemy, "attack", 0.35)
                enemy.attack_timer = enemy.effective_interval

                dealt = self._hit_hero(
                    target, enemy.effective_damage, enemy.magical
                )

                if EnemyAffix.VAMPIRIC in enemy.affixes and dealt > 0:
                    restored = min(enemy.max_hp - enemy.hp, dealt * 0.25)
                    enemy.hp += restored
                    if restored > 0:
                        self.damage_event.emit(DamageEvent(
                            enemy.enemy_id, enemy.x, enemy.y - 32,
                            restored, "heal",
                        ))

    def _party_snapshots(self):
        snapshots = []

        for snapshot in self.state.data.party:
            runtime = self._heroes.get(snapshot.hero_id)
            if runtime is None:
                snapshots.append(snapshot)
                continue

            # HeroState хранит совместимую HP-пропорцию.
            hp = (
                0 if not runtime.alive
                else max(
                    1,
                    min(
                        snapshot.max_hp,
                        math.ceil(
                            runtime.hp / runtime.stats.max_hp * snapshot.max_hp
                        ),
                    ),
                )
            )
            snapshots.append(replace(snapshot, hp=hp))

        return tuple(snapshots)

    def _progress(self) -> float:
        if self.phase == "between":
            return 1.0
        if self.phase != "battle":
            return 0.0

        remaining_hp = sum(enemy.hp for enemy in self._enemies)
        return max(0.0, min(1.0, 1.0 - remaining_hp / self._wave_total_hp))

    def flush(self) -> None:
        """Перед автосохранением: записать HP и агрегат прогресса волны."""
        self.state.assert_owner_thread()
        if not self._heroes:
            return

        before = self.state.data
        progress = self._progress() if self.phase == "battle" else 0.0
        progress = min(progress, 0.999999999)

        after = replace(
            before,
            party=self._party_snapshots(),
            wave_fraction=str(Fraction(str(progress))),
        )
        self.state.commit(before, after)

    def _finish_wave(self) -> None:
        # Фазу меняем до событий — повторная выдача награды невозможна.
        self.phase = "between"
        self._phase_left = BETWEEN_WAVES
        self._dots.clear()

        before = self.state.data
        profile = before.farm

        gold = math.floor(
            profile.gold_per_wave * (1 + Fraction(str(profile.gold_bonus)))
        )
        if self._wave % 10 == 0 and "gold_fever" in self._effects.keystones:
            gold *= 4

        xp = profile.exp_per_wave
        base, extra = divmod(xp, len(before.party))
        gains = [base] * len(before.party)

        for index in range(extra):
            gains[(before.exp_cursor + index) % len(gains)] += 1

        party = tuple(
            replace(hero, total_exp=hero.total_exp + gains[index])
            for index, hero in enumerate(self._party_snapshots())
        )

        stash = list(before.stash)
        if self.rng.random() < profile.drop_chance:
            rarity = self.rng.choices(
                [entry[0] for entry in C.LOOT_TABLE],
                weights=[entry[1] for entry in C.LOOT_TABLE],
                k=1,
            )[0]
            prices = {entry[0]: entry[2] for entry in C.LOOT_TABLE}
            price = prices[rarity]

            if len(stash) >= before.stash_capacity:
                gold += math.floor(
                    price * (1 + Fraction(str(profile.sale_bonus)))
                )
            else:
                item = generate_item(
                    1 + (self._wave - 1) // 100,
                    rarity,
                    source_wave=self._wave,
                    rng=self.rng,
                )
                known_ids = {entry.item_id for entry in stash}
                known_ids.update(
                    entry.item_id
                    for build in before.hero_builds
                    for entry in build.equipment
                )
                while item.item_id in known_ids:
                    item = replace(
                        item,
                        item_id=f"item-{self.rng.getrandbits(128):032x}",
                    )
                stash.append(replace(item, sell_value=price))

        # Частичная восстановленная волна не является корректным
        # образцом полной скорости зачистки.
        farm = profile
        if not self._continued_wave:
            measured = max(C.MIN_WAVE_CLEAR_SECONDS, self._wave_real_elapsed)
            farm = replace(
                profile,
                average_wave_clear_seconds=max(
                    C.MIN_WAVE_CLEAR_SECONDS,
                    profile.average_wave_clear_seconds * 0.85
                    + measured * 0.15,
                ),
            )

        after = replace(
            before,
            gold=before.gold + gold,
            party=party,
            stash=tuple(stash),
            cleared_waves=before.cleared_waves + 1,
            wave_fraction="0",
            exp_cursor=(before.exp_cursor + extra) % len(before.party),
            farm=farm,
        )

        self.state.commit(before, after)
        self.wave_finished.emit(self._wave)

    def _defeat(self, reason: str) -> None:
        self.phase = "defeat"
        self._phase_left = DEFEAT_DELAY
        self._dots.clear()

        before = self.state.data
        self.state.commit(
            before,
            replace(
                before,
                party=self._party_snapshots(),
                wave_fraction="0",
            ),
        )
        self.party_defeated.emit(reason)

    def frame(self) -> CombatFrame:
        actors = []

        for hero in self._heroes.values():
            actors.append(ActorFrame(
                actor_id=hero.hero.hero_id,
                sprite_key=hero.hero.hero_class.value,
                name=hero.hero.name,
                x=hero.x,
                y=hero.y,
                hp=hero.hp,
                max_hp=hero.stats.max_hp,
                animation=hero.animation,
                enemy=False,
                mana=hero.mana,
                max_mana=hero.stats.max_mana,
            ))

        for enemy in self._enemies:
            if not enemy.alive and enemy.corpse_left <= 0:
                continue

            actors.append(ActorFrame(
                actor_id=enemy.enemy_id,
                sprite_key=enemy.template_id,
                name=enemy.name,
                x=enemy.x,
                y=enemy.y,
                hp=enemy.hp,
                max_hp=enemy.max_hp,
                animation=enemy.animation,
                enemy=True,
                elite=bool(enemy.affixes),
                boss=enemy.is_boss,
                rage=enemy.rage / enemy.rage_max,
            ))

        boss = any(enemy.is_boss for enemy in self._enemies)

        return CombatFrame(
            wave=self._wave,
            act=act_for_wave(self._wave),
            phase=self.phase,
            time=self.time,
            progress=self._progress(),
            barrier=self.barrier,
            barrier_max=self.barrier_max,
            boss_remaining=(
                max(0.0, BOSS_TIME_LIMIT - self._wave_elapsed)
                if boss and self.phase == "battle" else None
            ),
            actors=tuple(actors),
        )
```

### Примечание к сохранению активного боя

`flush()` сохраняет HP и **агрегат выполненной работы волны**. После перезапуска враги генерируются заново и получают пропорционально уменьшенное здоровье.

Это согласовано с моделью Модуля 1, которая хранит дробь волны, а не позиции каждого моба. Точное восстановление DoT, RNG и кулдаунов между запусками потребует отдельной версии схемы сохранения.

```python
from __future__ import annotations

import math
from enum import Enum


class AnimationState(str, Enum):
    IDLE = "idle"
    RUN = "run"
    ATTACK = "attack"
    CAST = "cast"
    HURT = "hurt"
    DIE = "die"


LOOPING = {
    AnimationState.IDLE,
    AnimationState.RUN,
}

DURATIONS = {
    AnimationState.IDLE: 0.80,
    AnimationState.RUN: 0.50,
    AnimationState.ATTACK: 0.35,
    AnimationState.CAST: 0.45,
    AnimationState.HURT: 0.20,
    AnimationState.DIE: 0.60,
}


class AnimationController:
    def __init__(self, frame_count: int = 4) -> None:
        if type(frame_count) is not int or frame_count < 1:
            raise ValueError("frame_count должен быть целым >= 1")

        self.frame_count = frame_count
        self.state = AnimationState.IDLE
        self.elapsed = 0.0

    def set_state(self, state: AnimationState | str) -> bool:
        state = AnimationState(state)

        # Главное правило FSM: одинаковое состояние не сбрасывает кадр.
        if state == self.state:
            return False

        self.state = state
        self.elapsed = 0.0
        return True

    def advance(self, seconds: float) -> None:
        if not math.isfinite(seconds) or seconds < 0:
            raise ValueError("Некорректная дельта времени")

        duration = DURATIONS[self.state]

        if self.state in LOOPING:
            self.elapsed = (self.elapsed + seconds) % duration
        else:
            # Неповторяющиеся анимации удерживают последний кадр.
            # Переход обратно в IDLE задаёт боевой движок.
            self.elapsed = min(duration, self.elapsed + seconds)

    @property
    def frame_index(self) -> int:
        duration = DURATIONS[self.state]
        return min(
            self.frame_count - 1,
            int(self.elapsed / duration * self.frame_count),
        )

    @property
    def finished(self) -> bool:
        return (
            self.state not in LOOPING
            and self.elapsed >= DURATIONS[self.state]
        )
```

Здесь нет автоматического `DIE → IDLE`: мёртвая сущность удерживает последний кадр до удаления из визуального снимка.

Формат ассетов:

```text
assets/sprites/

│   ├── idle.png
│   ├── run.png
│   ├── attack.png
│   ├── cast.png
│   ├── hurt.png
│   └── die.png
├── assassin/
├── pyromancer/
├── ranger/
├── necromancer/
├── paladin/
├── skeleton/
└── ...
```

Каждый файл — **горизонтальная лента из четырёх кадров** одинаковой ширины. Размер исходных кадров произвольный, но ширина изображения должна делиться на четыре.

## `gfx/sprite_loader.py`

```python
from __future__ import annotations

import math
from pathlib import Path

from PySide6.QtCore import QPointF, QRectF, Qt
from PySide6.QtGui import QColor, QPainter, QPen, QPixmap, QPolygonF

import config as C
from gfx.animations import AnimationState


HERO_COLORS = {
    "knight": "#b8b1a3",
    "assassin": "#776195",
    "pyromancer": "#e16c37",
    "ranger": "#71935d",
    "necromancer": "#8b66b3",
    "paladin": "#d0b66c",
}


class SpriteLoader:
    def __init__(self, root: Path = C.SPRITES_DIR) -> None:
        self.root = Path(root)
        self._cache: dict[tuple, tuple[QPixmap, ...]] = {}

    def clear(self) -> None:
        self._cache.clear()

    def frames(
        self,
        key: str,
        state: AnimationState | str,
        *,
        size: int = 36,
        dpr: float = 1.0,
        frame_count: int = 4,
    ) -> tuple[QPixmap, ...]:
        state = AnimationState(state)

        if size < 1 or frame_count < 1:
            raise ValueError("Некорректный размер кадров")
        if not math.isfinite(dpr) or dpr <= 0:
            raise ValueError("DPR должен быть положительным")

        dpr = round(dpr, 3)
        cache_key = (key, state.value, size, dpr, frame_count)

        if cache_key in self._cache:
            return self._cache[cache_key]

        path = self.root / key / f"{state.value}.png"
        sheet = QPixmap(str(path)) if path.is_file() else QPixmap()
        result = []

        if (
            not sheet.isNull()
            and sheet.width() >= frame_count
            and sheet.width() % frame_count == 0
        ):
            width = sheet.width() // frame_count
            physical_size = max(1, math.ceil(size * dpr))

            for index in range(frame_count):
                frame = sheet.copy(index * width, 0, width, sheet.height())
                frame = frame.scaled(
                    physical_size,
                    physical_size,
                    Qt.AspectRatioMode.IgnoreAspectRatio,
                    Qt.TransformationMode.SmoothTransformation,
                )
                frame.setDevicePixelRatio(dpr)
                result.append(frame)
        else:
            for index in range(frame_count):
                result.append(
                    self._fallback(key, state, index, frame_count, size, dpr)
                )

        frames = tuple(result)
        self._cache[cache_key] = frames
        return frames

    @staticmethod
    def _fallback(
        key: str,
        state: AnimationState,
        index: int,
        count: int,
        size: int,
        dpr: float,
    ) -> QPixmap:
        physical_size = max(1, math.ceil(size * dpr))
        pixmap = QPixmap(physical_size, physical_size)
        pixmap.setDevicePixelRatio(dpr)
        pixmap.fill(Qt.GlobalColor.transparent)

        painter = QPainter(pixmap)
        painter.setRenderHint(QPainter.RenderHint.Antialiasing)
        painter.scale(size / 36.0, size / 36.0)

        progress = index / max(1, count - 1)
        color = QColor(HERO_COLORS.get(key, "#9b4549"))

        if state == AnimationState.HURT:
            color = color.lighter(165)

        if state == AnimationState.DIE:
            painter.setOpacity(1.0 - progress * 0.80)
            painter.translate(18, 31)
            painter.rotate(75 * progress)
            painter.translate(-18, -31)

        bob = (
            math.sin(index / count * math.tau) * 1.5
            if state in (AnimationState.IDLE, AnimationState.RUN)
            else 0.0
        )
        painter.translate(0, bob)

        painter.setPen(QPen(QColor("#17131d"), 1.2))
        painter.setBrush(color)

        painter.drawEllipse(QRectF(13, 5, 10, 10))

        body = QPolygonF([
            QPointF(12, 16),
            QPointF(24, 16),
            QPointF(27, 29),
            QPointF(9, 29),
        ])
        painter.drawPolygon(body)

        leg_offset = (
            math.sin(index / count * math.tau) * 3
            if state == AnimationState.RUN else 0
        )
        painter.setPen(QPen(color.darker(130), 4))
        painter.drawLine(QPointF(14, 26), QPointF(12 + leg_offset, 34))
        painter.drawLine(QPointF(22, 26), QPointF(24 - leg_offset, 34))

        painter.setPen(QPen(QColor("#dacda8"), 2))
        if state == AnimationState.ATTACK:
            painter.drawLine(
                QPointF(24, 18),
                QPointF(29 + progress * 6, 7 + progress * 16),
            )
        else:
            painter.drawLine(QPointF(26, 12), QPointF(27, 29))

        if state == AnimationState.CAST:
            glow = QColor("#bd73ed")
            glow.setAlpha(120 + int(progress * 100))
            painter.setPen(QPen(glow, 2))
            painter.setBrush(Qt.BrushStyle.NoBrush)
            radius = 3 + progress * 5
            painter.drawEllipse(QPointF(28, 12), radius, radius)

        painter.end()
        return pixmap
```

```python
import math
from dataclasses import dataclass

from PySide6.QtCore import QElapsedTimer, QPointF, QRectF, Qt, QTimer
from PySide6.QtGui import (

from PySide6.QtWidgets import QWidget

import config as C
from engine.combat_manager import CombatFrame, CombatManager, DamageEvent
from gfx.animations import AnimationController
from gfx.sprite_loader import SpriteLoader
from models.enemy import ACT_NAMES, Act


BIOME_COLORS = {
    Act.CRYPTS: ("#171322", "#282033", "#443444"),
    Act.FOREST: ("#101e1b", "#20352b", "#365043"),
    Act.CALDERA: ("#251318", "#492326", "#713832"),
    Act.CITADEL: ("#151722", "#292c42", "#41475e"),
}

TEXT_COLORS = {
    "physical": "#f0ece4",
    "magical": "#ba79ef",
    "heal": "#78d58c",
    "dodge": "#a1a1aa",
    "barrier": "#80b8df",
}


@dataclass
class FloatingText:
    x: float
    y: float
    text: str
    color: QColor
    age: float = 0.0
    lifetime: float = 0.85


class BattleStage(QWidget):
    """
    Единственный драйвер advance() для переданного CombatManager.

    Если движок уже обновляет другой контроллер, передайте drive_combat=False.
    """

    def __init__(
        self,
        combat: CombatManager,
        parent: QWidget | None = None,
        *,
        drive_combat: bool = True,
    ) -> None:
        super().__init__(parent)

        self.combat = combat
        self.drive_combat = drive_combat
        self.loader = SpriteLoader()

        self._frame = combat.frame()
        self._animations: dict[str, AnimationController] = {}
        self._texts: list[FloatingText] = []
        self._last_simulation_time = self._frame.time

        self.setFixedSize(C.BATTLE_STAGE_WIDTH, C.BATTLE_STAGE_HEIGHT)
        self.setAttribute(Qt.WidgetAttribute.WA_OpaquePaintEvent)
        self.setToolTip("Боевой театр Бездны")

        combat.frame_changed.connect(self._receive_frame)
        combat.damage_event.connect(self._receive_damage)

        self._clock = QElapsedTimer()
        self._clock.start()

        self._timer = QTimer(self)
        self._timer.setInterval(16)
        self._timer.timeout.connect(self._tick)
        self._timer.start()

    def _receive_frame(self, frame: CombatFrame) -> None:
        self._frame = frame

        ids = {actor.actor_id for actor in frame.actors}
        for actor_id in tuple(self._animations):
            if actor_id not in ids:
                del self._animations[actor_id]

        for actor in frame.actors:
            animator = self._animations.setdefault(
                actor.actor_id,
                AnimationController(),
            )
            animator.set_state(actor.animation)

    def _receive_damage(self, event: DamageEvent) -> None:
        if event.kind == "dodge":
            text = "УВОРОТ"
        elif event.kind == "barrier":
            text = "БАРЬЕР"
        elif event.kind == "heal":
            text = f"+{max(1, round(event.amount))}"
        else:
            text = str(max(1, round(event.amount)))

        color = QColor(
            C.PALETTE["gold"]
            if event.critical
            else TEXT_COLORS.get(event.kind, "#ffffff")
        )

        # Небольшое детерминированное смещение уменьшает наложение текста.
        offset = (len(self._texts) % 5 - 2) * 4
        self._texts.append(FloatingText(
            x=event.x + offset,
            y=event.y,
            text=text,
            color=color,
        ))

        # Жёсткий предел исключает рост памяти при массовом AoE.
        if len(self._texts) > 128:
            del self._texts[:-128]

    def _tick(self) -> None:
        real_dt = max(0.0, self._clock.restart() / 1000.0)

        # После перетаскивания окна/паузы отладчика не выполняем
        # многосекундную симуляцию одним GUI-кадром.
        dt = min(real_dt, 0.25)

        if self.drive_combat:
            self.combat.advance(dt)

        simulation_dt = max(
            0.0,
            self._frame.time - self._last_simulation_time,
        )
        self._last_simulation_time = self._frame.time

        for animator in self._animations.values():
            animator.advance(simulation_dt)

        alive_texts = []
        for text in self._texts:
            text.age += dt
            if text.age < text.lifetime:
                alive_texts.append(text)

        self._texts = alive_texts
        self.update()

    def _draw_background(self, painter: QPainter) -> None:
        base, far_color, near_color = BIOME_COLORS[self._frame.act]
        painter.fillRect(self.rect(), QColor(base))

        # Два слоя с разными скоростями.
        for layer, (color, speed, baseline, height) in enumerate((
            (far_color, 4.0, 91.0, 42.0),
            (near_color, 9.0, 121.0, 28.0),
        )):
            painter.setPen(Qt.PenStyle.NoPen)
            painter.setBrush(QColor(color))

            offset = (self._frame.time * speed) % 96

            for index in range(-1, 8):
                x = index * 96 - offset
                peak = height * (0.7 + 0.3 * math.sin(index * 2.3 + layer))

                if self._frame.act in (Act.CRYPTS, Act.CITADEL):
                    painter.drawRect(QRectF(x + 10, baseline - peak, 28, peak))
                    painter.drawPolygon(QPolygonF([
                        QPointF(x + 5, baseline - peak),
                        QPointF(x + 24, baseline - peak - 15),
                        QPointF(x + 43, baseline - peak),
                    ]))
                else:
                    painter.drawPolygon(QPolygonF([
                        QPointF(x - 15, baseline),
                        QPointF(x + 35, baseline - peak),
                        QPointF(x + 88, baseline),
                    ]))

        ground = QLinearGradient(0, 108, 0, 144)
        ground.setColorAt(0, QColor(18, 15, 23, 80))
        ground.setColorAt(1, QColor(10, 9, 15, 255))
        painter.fillRect(QRectF(0, 108, 520, 36), ground)

    def _draw_barrier(self, painter: QPainter) -> None:
        if self._frame.barrier_max <= 0:
            return

        ratio = self._frame.barrier / self._frame.barrier_max
        if ratio <= 0:
            return

        pulse = 0.85 + 0.15 * math.sin(self._frame.time * 3)
        rect = QRectF(40, 35, 168, 101)

        gradient = QRadialGradient(QPointF(124, 94), 87)
        gradient.setColorAt(0.0, QColor(100, 155, 225, 3))
        gradient.setColorAt(0.78, QColor(130, 125, 230, 12))
        gradient.setColorAt(
            1.0,
            QColor(136, 179, 245, int(65 * ratio * pulse)),
        )

        painter.setBrush(gradient)
        painter.setPen(QPen(
            QColor(148, 190, 255, int(150 * ratio * pulse)),
            1.3,
        ))
        painter.drawEllipse(rect)

    def _draw_actors(self, painter: QPainter) -> None:
        dpr = self.devicePixelRatioF()

        for actor in sorted(self._frame.actors, key=lambda entry: entry.y):
            size = 48 if actor.boss else 36
            animator = self._animations.get(actor.actor_id)
            if animator is None:
                animator = AnimationController()
                animator.set_state(actor.animation)
                self._animations[actor.actor_id] = animator

            frames = self.loader.frames(
                actor.sprite_key,
                actor.animation,
                size=size,
                dpr=dpr,
            )
            pixmap = frames[animator.frame_index % len(frames)]

            painter.save()
            painter.translate(actor.x, actor.y)

            if actor.enemy:
                painter.scale(-1, 1)

            painter.drawPixmap(
                QPointF(-size / 2, -size),
                pixmap,
            )
            painter.restore()

            if actor.hp <= 0:
                continue

            bar_width = 46 if actor.boss else 27
            left = actor.x - bar_width / 2
            top = actor.y - size - 5

            painter.fillRect(
                QRectF(left, top, bar_width, 3),
                QColor("#27232e"),
            )
            painter.fillRect(
                QRectF(
                    left, top,
                    bar_width * max(0.0, min(1.0, actor.hp / actor.max_hp)),
                    3,
                ),
                QColor("#bd5252" if actor.enemy else "#72aa85"),
            )

            if actor.elite:
                painter.setPen(QColor(C.PALETTE["gold"]))
                painter.drawText(QPointF(actor.x - 4, top - 2), "◆")

            if actor.boss:
                painter.fillRect(
                    QRectF(left, top + 5, bar_width, 2),
                    QColor("#352324"),
                )
                painter.fillRect(
                    QRectF(left, top + 5, bar_width * actor.rage, 2),
                    QColor("#ef853d"),
                )

    def _draw_texts(self, painter: QPainter) -> None:
        painter.setFont(QFont("Sans Serif", 8, QFont.Weight.Bold))

        for text in self._texts:
            alpha = max(0.0, 1.0 - text.age / text.lifetime)
            color = QColor(text.color)
            color.setAlphaF(alpha)

            y = text.y - text.age * 27
            rect = QRectF(text.x - 45, y - 10, 90, 20)

            shadow = QColor(0, 0, 0, int(180 * alpha))
            painter.setPen(shadow)
            painter.drawText(
                rect.translated(1, 1),
                Qt.AlignmentFlag.AlignCenter,
                text.text,
            )

            painter.setPen(color)
            painter.drawText(rect, Qt.AlignmentFlag.AlignCenter, text.text)

    def _draw_hud(self, painter: QPainter) -> None:
        painter.fillRect(QRectF(0, 0, 520, 24), QColor(10, 9, 15, 210))
        painter.setFont(QFont("Sans Serif", 8))
        painter.setPen(QColor(C.PALETTE["text"]))

        title = f"{ACT_NAMES[self._frame.act]}  ·  Волна {self._frame.wave}"
        painter.drawText(QPointF(9, 15), title)

        if self._frame.boss_remaining is not None:
            painter.setPen(QColor(C.PALETTE["gold"]))
            painter.drawText(
                QRectF(335, 2, 175, 19),
                Qt.AlignmentFlag.AlignRight | Qt.AlignmentFlag.AlignVCenter,
                f"БОСС  {self._frame.boss_remaining:04.1f} с",
            )
        elif self._frame.phase == "defeat":
            painter.setPen(QColor("#dc7777"))
            painter.drawText(QPointF(326, 15), "Отступление…")
        elif self._frame.phase == "between":
            painter.drawText(QPointF(347, 15), "Волна отражена")

        painter.fillRect(QRectF(8, 137, 504, 3), QColor("#302635"))
        painter.fillRect(
            QRectF(8, 137, 504 * self._frame.progress, 3),
            QColor(C.PALETTE["gold"]),
        )

        painter.setPen(QPen(QColor(C.PALETTE["border"]), 1))
        painter.setBrush(Qt.BrushStyle.NoBrush)
        painter.drawRect(QRectF(0.5, 0.5, 519, 143))

    def paintEvent(self, event) -> None:
        painter = QPainter(self)
        painter.setRenderHint(QPainter.RenderHint.Antialiasing)
        painter.setClipRect(self.rect())

        self._draw_background(painter)
        self._draw_barrier(painter)
        self._draw_actors(painter)
        self._draw_texts(painter)
        self._draw_hud(painter)

        painter.end()
```

### Управление памятью

- аниматоры удаляются, когда идентификатор исчезает из снимка;
- всплывающие цифры живут максимум 0,85 секунды;
- одновременно хранится максимум 128 текстов;
- кэш спрайтов разделяется между всеми мобами одного вида;
- динамические мобы не являются `QObject`, поэтому для них не нужны `hide()` и `deleteLater()`.

Таймер виджета имеет Qt-родителя и уничтожается вместе с виджетом.

```python
    def test_partial_progress_is_flushed(self) -> None:

    def test_healing_is_capped(self) -> None:

    def test_all_level_fifteen_skill_effects_are_supported(self) -> None:
```

# 8. Запускаемый стенд

Это отдельная точка входа: текущий `main.py` Модуля 1 можно сохранить для диагностики.

## `main_battle.py`

```python
from __future__ import annotations

import signal
import sys

from PySide6.QtCore import QLockFile, QTimer
from PySide6.QtWidgets import QApplication

import config as C
from engine.combat_manager import CombatManager
from engine.offline_manager import OfflineManager
from engine.save_manager import SaveError, SaveManager
from engine.state import GameState
from ui.battle_stage import BattleStage


def main() -> int:
    app = QApplication(sys.argv)
    app.setApplicationName(C.APP_NAME)
    app.setOrganizationName(C.ORGANIZATION_NAME)

    try:
        C.DATA_DIR.mkdir(parents=True, exist_ok=True)
    except OSError as exc:
        print(f"Ошибка каталога данных: {exc}", file=sys.stderr)
        return 1

    lock = QLockFile(str(C.LOCK_PATH))
    lock.setStaleLockTime(0)

    if not lock.tryLock(0):
        print(
            "Не удалось получить блокировку. "
            "Закройте другой экземпляр Desktop Abyss.",
            file=sys.stderr,
        )
        return 1

    try:
        state = GameState.instance()
        saves = SaveManager()

        try:
            state.load(saves.load())
            offline = OfflineManager().claim(state, saves)
        except (SaveError, ValueError) as exc:
            print(f"Ошибка загрузки: {exc}", file=sys.stderr)
            return 1

        print(
            f"Оффлайн: {offline.waves_cleared} волн, "
            f"+{offline.total_gold} золота, +{offline.total_exp} XP"
        )

        combat = CombatManager(state)

        try:
            combat.start()
        except ValueError as exc:
            print(f"Ошибка состава отряда: {exc}", file=sys.stderr)
            return 1

        stage = BattleStage(combat)
        stage.setWindowTitle(f"{C.APP_NAME} — Battle Stage")
        stage.show()

        exit_status = 0

        def checkpoint() -> None:
            nonlocal exit_status
            try:
                combat.flush()
                saves.checkpoint(state)
            except (SaveError, ValueError) as exc:
                exit_status = 1
                print(f"Ошибка сохранения: {exc}", file=sys.stderr)

        autosave = QTimer(app)
        autosave.setInterval(C.AUTOSAVE_INTERVAL_MS)
        autosave.timeout.connect(checkpoint)
        autosave.start()

        interrupt_timer = QTimer(app)
        interrupt_timer.setInterval(200)
        interrupt_timer.timeout.connect(lambda: None)
        interrupt_timer.start()

        signal.signal(signal.SIGINT, lambda *_: app.quit())
        signal.signal(signal.SIGTERM, lambda *_: app.quit())

        app.aboutToQuit.connect(checkpoint)

        qt_status = app.exec()
        return qt_status or exit_status

    finally:
        lock.unlock()


if __name__ == "__main__":
    raise SystemExit(main())
```

### Важные границы реализации

- `BattleStage` — готовый нижний боевой виджет, **не полное готическое окно приложения**.
- Визуальная симуляция использует условные дистанции атаки, а не физический движок столкновений.
- При длительной блокировке GUI виджет ограничивает догоняющий шаг до 250 мс; отдельный учёт сна системы ещё не реализован.
- Навык призыва наносит периодический урон от имени владельца; отдельный спрайт призванного существа пока не рисуется.
- Активный бой сохраняется каждые 30 секунд и при закрытии, поэтому аварийное завершение может потерять последний интервал — как в Модуле 1.

Я не выполнял этот код в вашей среде; приложенные тесты и стенд предназначены для следующего интеграционного прогона.
